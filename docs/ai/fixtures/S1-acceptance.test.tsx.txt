/** Independent P03 acceptance: required semantics and focus after invalid/400 submissions. */
import { describe, expect, it } from 'vitest'
import { screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { I18nextProvider } from 'react-i18next'
import { http, HttpResponse } from 'msw'
import { createAppI18n } from '../lib/i18n'
import { renderWithProviders } from './renderWithProviders'
import { server } from './server'
import { ArticlesPage } from '../features/articles/components/ArticlesPage'

const base = import.meta.env.VITE_API_BASE_URL as string

describe.each(['en', 'uk'])('independent P03 accessibility acceptance (%s)', (language) => {
  /** Render the actual page in an explicit locale; test-local provider state only. */
  function renderPage() {
    const instance = createAppI18n(language)
    renderWithProviders(<I18nextProvider i18n={instance}><ArticlesPage /></I18nextProvider>)
    return instance.getFixedT(language, 'articles')
  }

  it('announces required state and focuses the invalid title after submit', async () => {
    const t = renderPage()
    const user = userEvent.setup()
    const title = screen.getByRole('textbox', { name: t('titleLabel') })
    expect(title).toBeRequired()
    await user.click(screen.getByRole('button', { name: t('add') }))
    expect(await screen.findByText(t('titleRequired'))).toBeVisible()
    expect(title).toHaveAttribute('aria-invalid', 'true')
    expect(title).toHaveAccessibleDescription(t('titleRequired'))
    expect(title).toHaveFocus()
  })

  it('focuses the first server-invalid field and keeps submitted input', async () => {
    server.use(http.post(`${base}/api/v1/articles`, () => HttpResponse.json({errors: [
      {field: 'title', code: 'invalid', message: 'Choose another title'},
      {field: 'body', code: 'invalid', message: 'Revise the body'},
    ]}, {status: 400})))
    const t = renderPage()
    const user = userEvent.setup()
    const title = screen.getByRole('textbox', { name: t('titleLabel') })
    const body = screen.getByRole('textbox', { name: t('bodyLabel') })
    await user.type(title, '  Keep this title  ')
    await user.type(body, '  Keep this body  ')
    await user.click(screen.getByRole('button', { name: t('add') }))
    expect(await screen.findByText('Choose another title')).toBeVisible()
    expect(title).toHaveAccessibleDescription('Choose another title')
    expect(body).toHaveAccessibleDescription('Revise the body')
    expect(title).toHaveValue('  Keep this title  ')
    expect(body).toHaveValue('  Keep this body  ')
    expect(title).toHaveFocus()
  })
})
